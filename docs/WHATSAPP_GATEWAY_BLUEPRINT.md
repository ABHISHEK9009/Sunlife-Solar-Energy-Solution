# Standalone WhatsApp Gateway Blueprint

This guide provides the complete, production-grade blueprint to build, deploy, and connect this standalone WhatsApp Gateway in **any project** (Next.js, React, Node.js, Express, Django, Laravel, etc.).

---

## 1. Architecture Overview

```
┌─────────────────────────┐          REST API          ┌───────────────────────────┐
│     Your Main App       │  ───────────────────────>  │     WhatsApp Gateway      │
│  (Next.js / Express /   │   (with API Key Header)    │     (Node.js + Express    │
│   Laravel / Django)     │  <───────────────────────  │     + Baileys Engine)     │
└─────────────────────────┘                            └─────────────┬─────────────┘
             │                                                       │ WebSocket
      User scans QR                                                  │
             ▼                                                       ▼
      Admin Dashboard                                          WhatsApp Servers
```

### Why this design is ideal:
* **Free & No Meta Approval**: Uses `@whiskeysockets/baileys` (WhatsApp Web multi-device protocol). Any phone number can be linked by scanning a QR code.
* **Crash & Memory Isolation**: Baileys keeps a persistent WebSocket connection and writes auth credentials to disk. Running it in a separate process prevents crashes or memory spikes from affecting your main app.
* **Deploy Anywhere**: The gateway can run on the same VPS via PM2, or separately on a service like Railway, Render, or Docker with a persistent disk.

## 2. Step 1: Create the Gateway Folder & Install Dependencies

In your terminal, create a dedicated folder for the gateway:

```bash
mkdir whatsapp-gateway
cd whatsapp-gateway
npm init -y
```

Update your `package.json` to enable ES modules:
```json
{
  "name": "whatsapp-gateway",
  "version": "1.0.0",
  "type": "module",
  "scripts": {
    "start": "node server.js",
    "dev": "node --watch server.js"
  }
}
```

Install the required packages:
```bash
npm install @whiskeysockets/baileys @hapi/boom express qrcode pino dotenv
```

### 2. `baileys-manager.js`
This file handles the WhatsApp connection, session storage, QR code generation, and sending messages:

```javascript
import makeWASocket, {
  useMultiFileAuthState,
  DisconnectReason,
  fetchLatestBaileysVersion,
} from "@whiskeysockets/baileys";
import QRCode from "qrcode";
import pino from "pino";
import path from "path";

const SESSIONS_DIR = path.join(process.cwd(), "sessions");
const logger = pino({ level: "warn" });

let sock = null;
let connectionState = "DISCONNECTED"; // DISCONNECTED | CONNECTING | QR_READY | CONNECTED
let currentQrDataUrl = null;

export function getStatus() {
  if (connectionState === "DISCONNECTED") {
    startConnection().catch(console.error);
  }
  return {
    state: connectionState,
    connected: connectionState === "CONNECTED",
    qrAvailable: !!currentQrDataUrl,
  };
}

export function getQrCode() {
  return currentQrDataUrl;
}

export async function startConnection() {
  connectionState = "CONNECTING";
  currentQrDataUrl = null;

  // Persists login credentials to ./sessions folder
  const { state, saveCreds } = await useMultiFileAuthState(SESSIONS_DIR);
  const { version } = await fetchLatestBaileysVersion();

  sock = makeWASocket({
    version,
    auth: state,
    logger,
    browser: ["Custom CRM", "Chrome", "1.0.0"],
    connectTimeoutMs: 60000,
  });

  sock.ev.on("creds.update", saveCreds);

  sock.ev.on("connection.update", async (update) => {
    const { connection, lastDisconnect, qr } = update;

    // When a QR code is generated by WhatsApp
    if (qr) {
      currentQrDataUrl = await QRCode.toDataURL(qr, { width: 300, margin: 2 });
      connectionState = "QR_READY";
      console.log("[WhatsApp] QR Code ready for scanning.");
    }

    // Connection closed/lost
    if (connection === "close") {
      currentQrDataUrl = null;
      const statusCode = lastDisconnect?.error?.output?.statusCode;
      const shouldReconnect = statusCode !== DisconnectReason.loggedOut;

      connectionState = "DISCONNECTED";
      console.log(`[WhatsApp] Disconnected (code: ${statusCode}). Reconnecting: ${shouldReconnect}`);

      if (shouldReconnect) {
        setTimeout(startConnection, 5000);
      }
    } else if (connection === "open") {
      connectionState = "CONNECTED";
      currentQrDataUrl = null;
      console.log("[WhatsApp] Successfully connected!");
    }
  });
}

function formatJid(recipient) {
  const clean = recipient.toString().replace(/\D/g, "");
  // Formats: '919876543210@s.whatsapp.net' for individuals or '@g.us' for groups
  if (recipient.includes("@g.us")) return recipient;
  return `${clean}@s.whatsapp.net`;
}

export async function sendText(to, text) {
  if (!sock || connectionState !== "CONNECTED") {
    throw new Error("WhatsApp gateway is not connected.");
  }
  const jid = formatJid(to);
  return await sock.sendMessage(jid, { text });
}

export async function sendMedia(to, base64Data, filename, caption = "") {
  if (!sock || connectionState !== "CONNECTED") {
    throw new Error("WhatsApp gateway is not connected.");
  }
  const jid = formatJid(to);
  const buffer = Buffer.from(base64Data, "base64");
  return await sock.sendMessage(jid, {
    image: buffer,
    caption,
    fileName: filename || "file.jpg",
  });
}
```

### 3. `server.js`
This file creates the Express HTTP API that your other project calls:

```javascript
import express from "express";
import dotenv from "dotenv";
import { getStatus, getQrCode, sendText, sendMedia, startConnection } from "./baileys-manager.js";

dotenv.config();

const app = express();
app.use(express.json({ limit: "50mb" })); // Supports sending media

const API_KEY = process.env.WHATSAPP_GATEWAY_API_KEY || "secret";

// API Key Authentication Middleware
app.use((req, res, next) => {
  if (req.path === "/health") return next();
  const clientKey = req.headers["x-api-key"];
  if (!clientKey || clientKey !== API_KEY) {
    return res.status(401).json({ error: "Unauthorized: Invalid API Key" });
  }
  next();
});

// 1. Health check
app.get("/health", (req, res) => res.json({ status: "ok" }));

// 2. Check connection status
app.get("/status", (req, res) => res.json(getStatus()));

// 3. Get Base64 QR code image
app.get("/qr", (req, res) => {
  const qrCode = getQrCode();
  res.json({ success: !!qrCode, qrCode });
});

// 4. Send plain text message
app.post("/send-text", async (req, res) => {
  try {
    const { to, content } = req.body;
    if (!to || !content) return res.status(400).json({ error: "Missing 'to' or 'content'" });
    const result = await sendText(to, content);
    res.json({ success: true, result });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

// 5. Send image / media
app.post("/send-media", async (req, res) => {
  try {
    const { to, mediaBase64, filename, caption } = req.body;
    if (!to || !mediaBase64) return res.status(400).json({ error: "Missing 'to' or 'mediaBase64'" });
    const result = await sendMedia(to, mediaBase64, filename, caption);
    res.json({ success: true, result });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

const PORT = process.env.PORT || 8090;
app.listen(PORT, () => {
  console.log(`WhatsApp Gateway running on port ${PORT}`);
  startConnection();
});
```

## 4. Step 3: How to Use It from Your Other Project

In your other project (Next.js, Express, Django, Laravel, etc.):

### 1. Add Gateway URL to your other project's `.env`:
```ini
WHATSAPP_GATEWAY_URL=http://localhost:8090
WHATSAPP_GATEWAY_API_KEY=your_secret_api_key_here
```

### 2. Create a client helper in your other project:
```javascript
const GATEWAY_URL = process.env.WHATSAPP_GATEWAY_URL || "http://localhost:8090";
const API_KEY = process.env.WHATSAPP_GATEWAY_API_KEY;

// Send a message
export async function sendWhatsApp(phoneNumber, message) {
  // Always include country code without + or spaces (e.g., 919876543210)
  const cleanNumber = phoneNumber.replace(/\D/g, "");

  const response = await fetch(`${GATEWAY_URL}/send-text`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": API_KEY,
    },
    body: JSON.stringify({
      to: cleanNumber,
      content: message,
    }),
  });

  return await response.json();
}

// Check status
export async function checkWhatsAppStatus() {
  const response = await fetch(`${GATEWAY_URL}/status`, {
    headers: { "x-api-key": API_KEY },
  });
  return await response.json();
}

// Fetch QR Code for Frontend
export async function fetchWhatsAppQr() {
  const response = await fetch(`${GATEWAY_URL}/qr`, {
    headers: { "x-api-key": API_KEY },
  });
  return await response.json();
}
```

## 5. Step 4: Displaying the QR Code in your Frontend

In your other project's UI (React/Next.js/Vue):

```jsx
import { useEffect, useState } from "react";

export default function WhatsAppConnect() {
  const [qrCode, setQrCode] = useState(null);
  const [status, setStatus] = useState("DISCONNECTED");

  useEffect(() => {
    const interval = setInterval(async () => {
      const res = await fetch("/api/whatsapp/status").then((r) => r.json());
      setStatus(res.state);

      if (res.state === "QR_READY") {
        const qrRes = await fetch("/api/whatsapp/qr").then((r) => r.json());
        setQrCode(qrRes.qrCode);
      } else if (res.connected) {
        setQrCode(null); // connected, hide QR
      }
    }, 4000);

    return () => clearInterval(interval);
  }, []);

  return (
    <div>
      <h3>WhatsApp Status: {status}</h3>
      {status === "QR_READY" && qrCode && (
        <div>
          <p>Scan this QR code with WhatsApp on your phone:</p>
          <img src={qrCode} alt="WhatsApp QR Code" />
        </div>
      )}
      {status === "CONNECTED" && <p style={{ color: "green" }}>Connected & Ready!</p>}
    </div>
  );
}
```

## 6. Step 5: How to Run & Deploy in Production

1. **Keep it alive with PM2** (on any VPS/Linux server):
   ```bash
   npm install -g pm2
   cd whatsapp-gateway
   pm2 start server.js --name "whatsapp-gateway"
   pm2 startup
   pm2 save
   ```
2. **Session Persistence**:
   The login session is stored in the `whatsapp-gateway/sessions/` directory. As long as this folder is preserved, your WhatsApp account **remains linked even across server reboots**.
3. **Anti-Ban Rule**:
   Always add an **8 to 12-second delay** between outgoing messages if you are sending bulk/batch messages so WhatsApp does not flag your number.
